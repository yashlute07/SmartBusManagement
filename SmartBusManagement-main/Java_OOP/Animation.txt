import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.module.js";

// 1. Get HTML Elements
const container = document.getElementById("busAnimation3D");
const pauseButton = document.getElementById("busAnimPause");
const speedSlider = document.getElementById("busAnimSpeed");
const status = document.getElementById("busAnimStatus");

if (!container) {
    throw new Error("Bus animation container not found.");
}

// 2. Create Scene
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x9bd3f5);

// 3. Create Camera
const camera = new THREE.PerspectiveCamera(
    60,
    container.clientWidth / container.clientHeight,
    0.1,
    1000
);

camera.position.set(8, 7, 12);
camera.lookAt(0, 1, 0);

// 4. Create Renderer
const renderer = new THREE.WebGLRenderer({ antialias: true });

renderer.setSize(container.clientWidth, container.clientHeight);
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
renderer.shadowMap.enabled = true;

container.appendChild(renderer.domElement);

// 5. Add Lighting
const ambientLight = new THREE.AmbientLight(0xffffff, 1.2);
scene.add(ambientLight);

const directionalLight = new THREE.DirectionalLight(0xffffff, 2);
directionalLight.position.set(5, 10, 5);
directionalLight.castShadow = true;

scene.add(directionalLight);

// 6. Create Ground
const ground = new THREE.Mesh(
    new THREE.PlaneGeometry(200, 30),
    new THREE.MeshStandardMaterial({ color: 0x4b8b45 })
);

ground.rotation.x = -Math.PI / 2;
ground.position.y = -0.1;
ground.receiveShadow = true;

scene.add(ground);

// 7. Create Road
const road = new THREE.Mesh(
    new THREE.PlaneGeometry(200, 10),
    new THREE.MeshStandardMaterial({ color: 0x30343b })
);

road.rotation.x = -Math.PI / 2;
road.position.y = 0;
road.receiveShadow = true;

scene.add(road);

// 8. Create Road Lane Markings
const laneMarks = [];

for (let i = 0; i < 30; i++) {

    const mark = new THREE.Mesh(
        new THREE.BoxGeometry(2, 0.04, 0.15),
        new THREE.MeshStandardMaterial({ color: 0xffffff })
    );

    mark.position.set(-90 + i * 6, 0.04, 0);

    scene.add(mark);
    laneMarks.push(mark);
}

// 9. Create Bus Group
const bus = new THREE.Group();

// Bus Main Body
const busBody = new THREE.Mesh(
    new THREE.BoxGeometry(6, 2.5, 2.5),
    new THREE.MeshStandardMaterial({ color: 0x1565c0 })
);

busBody.position.y = 1.8;
busBody.castShadow = true;

bus.add(busBody);

// Bus Roof
const busRoof = new THREE.Mesh(
    new THREE.BoxGeometry(5.8, 0.3, 2.4),
    new THREE.MeshStandardMaterial({ color: 0xffffff })
);

busRoof.position.y = 3.2;
bus.add(busRoof);

// 10. Create Bus Windows
for (let i = 0; i < 5; i++) {

    const windowMesh = new THREE.Mesh(
        new THREE.BoxGeometry(0.85, 0.8, 0.08),
        new THREE.MeshStandardMaterial({
            color: 0x87ceeb,
            metalness: 0.2,
            roughness: 0.2
        })
    );

    windowMesh.position.set(
        -2.1 + i * 1.05,
        2.2,
        1.27
    );

    bus.add(windowMesh);
}

// 11. Create Bus Front
const front = new THREE.Mesh(
    new THREE.BoxGeometry(0.1, 1.4, 2.2),
    new THREE.MeshStandardMaterial({ color: 0x0d47a1 })
);

front.position.set(3.05, 1.8, 0);
bus.add(front);

// 12. Create Headlights
for (let i = 0; i < 2; i++) {

    const headlight = new THREE.Mesh(
        new THREE.BoxGeometry(0.12, 0.35, 0.45),
        new THREE.MeshStandardMaterial({
            color: 0xffff99,
            emissive: 0xffff33,
            emissiveIntensity: 0.5
        })
    );

    headlight.position.set(
        3.12,
        1.5,
        i === 0 ? -0.8 : 0.8
    );

    bus.add(headlight);
}

// 13. Create Wheels
const wheels = [];

for (let i = 0; i < 4; i++) {

    const wheel = new THREE.Mesh(
        new THREE.CylinderGeometry(0.55, 0.55, 0.25, 32),
        new THREE.MeshStandardMaterial({ color: 0x151515 })
    );

    wheel.rotation.x = Math.PI / 2;

    wheel.position.set(
        i < 2 ? -2 : 2,
        0.65,
        i % 2 === 0 ? -1.3 : 1.3
    );

    wheel.castShadow = true;

    bus.add(wheel);
    wheels.push(wheel);
}

// 14. Add Bus to Scene
bus.position.set(-2, 0, 0);
scene.add(bus);

// 15. Create Roadside Trees
const trees = [];

for (let i = 0; i < 20; i++) {

    const tree = new THREE.Group();

    const trunk = new THREE.Mesh(
        new THREE.CylinderGeometry(0.15, 0.2, 1.5, 8),
        new THREE.MeshStandardMaterial({ color: 0x754321 })
    );

    trunk.position.y = 0.75;
    tree.add(trunk);

    const leaves = new THREE.Mesh(
        new THREE.SphereGeometry(0.9, 16, 16),
        new THREE.MeshStandardMaterial({ color: 0x16803c })
    );

    leaves.position.y = 2;
    tree.add(leaves);

    tree.position.set(
        -80 + i * 9,
        0,
        i % 2 === 0 ? -8 : 8
    );

    scene.add(tree);
    trees.push(tree);
}

// 16. Animation Controls
let isPaused = false;
let speed = 1;

pauseButton.addEventListener("click", () => {

    isPaused = !isPaused;

    pauseButton.textContent = isPaused ? "Resume" : "Pause";
    status.textContent = isPaused ? "Bus is Paused" : "Bus is Moving";
});

speedSlider.addEventListener("input", () => {

    speed = Number(speedSlider.value);
});

// 17. Animation Loop
function animate() {

    requestAnimationFrame(animate);

    if (!isPaused) {

        // Move Bus
        bus.position.x += 0.08 * speed;

        // Reset Bus Position
        if (bus.position.x > 15) {
            bus.position.x = -15;
        }

        // Rotate Wheels
        wheels.forEach(wheel => {
            wheel.rotation.z -= 0.08 * speed;
        });

        // Move Road Markings
        laneMarks.forEach(mark => {

            mark.position.x -= 0.08 * speed;

            if (mark.position.x < -100) {
                mark.position.x = 100;
            }
        });

        // Move Trees
        trees.forEach(tree => {

            tree.position.x -= 0.08 * speed;

            if (tree.position.x < -100) {
                tree.position.x = 100;
            }
        });
    }

    renderer.render(scene, camera);
}

// 18. Responsive Window Resize
new ResizeObserver(() => {

    const width = container.clientWidth;
    const height = container.clientHeight;

    if (!width || !height) {
        return;
    }

    camera.aspect = width / height;
    camera.updateProjectionMatrix();

    renderer.setSize(width, height);

}).observe(container);

// 19. Start Animation
animate();
