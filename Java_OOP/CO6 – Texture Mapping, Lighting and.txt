CO6 – Texture Mapping, Lighting and Colour


Concept	Code used in code :

3D Objects	BoxGeometry, PlaneGeometry
Texture Mapping	CanvasTexture() वापरून checkerboard floor
Lighting	AmbientLight, DirectionalLight
Colour	Green, Red, Blue seats
Material	MeshStandardMaterial()
Shadow	castShadow, receiveShadow
Camera & Viewing	PerspectiveCamera()
Animation/Rendering	animate(), requestAnimationFrame()





import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.module.js";

const container = document.getElementById("bus3d");
const status = document.getElementById("seatStatus");

if (!container) {
    throw new Error("3D bus container not found.");
}

// 1. Scene Setup
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x101827);

// 2. Camera Setup
const camera = new THREE.PerspectiveCamera(
    48,
    container.clientWidth / container.clientHeight,
    0.1,
    100
);

camera.position.set(0, 19, 13);
camera.up.set(0, 0, -1);
camera.lookAt(0, 0, 0);

// 3. Renderer Setup
const renderer = new THREE.WebGLRenderer({ antialias: true });

renderer.setSize(container.clientWidth, container.clientHeight);
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
renderer.shadowMap.enabled = true;

container.appendChild(renderer.domElement);

// 4. Lighting
scene.add(new THREE.AmbientLight(0xffffff, 1.1));

const light = new THREE.DirectionalLight(0xffffff, 2);
light.position.set(-5, 12, -5);
light.castShadow = true;

scene.add(light);

// 5. Checkerboard Floor Texture
const textureCanvas = document.createElement("canvas");

textureCanvas.width = 256;
textureCanvas.height = 256;

const ctx = textureCanvas.getContext("2d");

for (let y = 0; y < 8; y++) {
    for (let x = 0; x < 8; x++) {

        ctx.fillStyle = (x + y) % 2 === 0
            ? "#9ba5aa"
            : "#5f6e78";

        ctx.fillRect(x * 32, y * 32, 32, 32);
    }
}

const floorTexture = new THREE.CanvasTexture(textureCanvas);

floorTexture.wrapS = THREE.RepeatWrapping;
floorTexture.wrapT = THREE.RepeatWrapping;
floorTexture.repeat.set(2, 5);

const floor = new THREE.Mesh(
    new THREE.PlaneGeometry(6, 18),
    new THREE.MeshStandardMaterial({
        map: floorTexture,
        roughness: 0.9
    })
);

floor.rotation.x = -Math.PI / 2;
floor.position.y = 0;
floor.receiveShadow = true;

scene.add(floor);

// 6. Function to Create 3D Box Objects
function addBox(x, y, z, sx, sy, sz, color) {

    const mesh = new THREE.Mesh(
        new THREE.BoxGeometry(sx, sy, sz),
        new THREE.MeshStandardMaterial({
            color,
            roughness: 0.65
        })
    );

    mesh.position.set(x, y, z);
    mesh.castShadow = true;
    mesh.receiveShadow = true;

    scene.add(mesh);

    return mesh;
}

// 7. Bus Side Walls and End Walls
addBox(-3.1, 0.7, 0, 0.25, 1.4, 18, 0x265980);
addBox(3.1, 0.7, 0, 0.25, 1.4, 18, 0x265980);

addBox(0, 0.7, -8.8, 6.2, 1.4, 0.3, 0x326699);
addBox(0, 0.7, 8.8, 6.2, 1.4, 0.3, 0x326699);

// 8. Central Aisle and Driver Area
addBox(0, 0.05, 0, 0.65, 0.08, 17, 0xd9c9a0);
addBox(0, 0.4, -8.1, 1.5, 0.7, 0.9, 0x444b5a);

// 9. Create 20 Seats (10 Rows x 2 Columns)
const seats = [];
let selectedSeat = 0;

function seatColor(index) {

    if (seats[index]?.userData.booked) {
        return 0xdc2626;
    }

    if (index === selectedSeat) {
        return 0x2563eb;
    }

    return 0x16a34a;
}

// 10. Update Seat Colours and Status
function updateSeats() {

    seats.forEach((seat, index) => {
        seat.material.color.setHex(seatColor(index));
    });

    const booked = seats.filter(
        seat => seat.userData.booked
    ).length;

    status.textContent =
        `Selected Seat: ${selectedSeat + 1} | Booked: ${booked} | Vacant: ${20 - booked}`;
}

// 11. Create Seats and Seat Components
for (let i = 0; i < 20; i++) {

    const x = i % 2 === 0 ? -1.25 : 1.25;
    const z = -7.5 + Math.floor(i / 2) * 1.65;

    const seat = addBox(
        x, 0.2, z,
        1, 0.35, 1,
        0x16a34a
    );

    seat.userData.index = i;
    seat.userData.booked = false;

    seats.push(seat);

    // Seat Backrest
    addBox(x, 0.65, z + 0.35, 1, 0.8, 0.2, 0x64748b);

    // Left Armrest
    addBox(x - 0.48, 0.42, z, 0.08, 0.3, 0.8, 0x475569);

    // Right Armrest
    addBox(x + 0.48, 0.42, z, 0.08, 0.3, 0.8, 0x475569);
}

// 12. Mouse-Based Seat Selection
const raycaster = new THREE.Raycaster();
const pointer = new THREE.Vector2();

renderer.domElement.addEventListener("click", event => {

    const rect = renderer.domElement.getBoundingClientRect();

    pointer.x =
        ((event.clientX - rect.left) / rect.width) * 2 - 1;

    pointer.y =
        -((event.clientY - rect.top) / rect.height) * 2 + 1;

    raycaster.setFromCamera(pointer, camera);

    const hits = raycaster.intersectObjects(seats);

    if (hits.length) {

        selectedSeat = hits[0].object.userData.index;

        updateSeats();
    }
});

// 13. Book Seat Function
function bookSeat() {

    seats[selectedSeat].userData.booked = true;

    updateSeats();
}

// 14. Vacate Seat Function
function vacateSeat() {

    seats[selectedSeat].userData.booked = false;

    updateSeats();
}

// 15. Reset Seats Function
function resetSeats() {

    seats.forEach(seat => {
        seat.userData.booked = false;
    });

    selectedSeat = 0;

    updateSeats();
}

// 16. Connect Buttons with Functions
document.getElementById("book3dSeat")
    .addEventListener("click", bookSeat);

document.getElementById("vacate3dSeat")
    .addEventListener("click", vacateSeat);

document.getElementById("reset3dSeats")
    .addEventListener("click", resetSeats);

// 17. Keyboard Controls
window.addEventListener("keydown", event => {

    if (
        event.key === "ArrowRight" ||
        event.key === "ArrowDown"
    ) {
        selectedSeat = (selectedSeat + 1) % 20;
    }

    else if (
        event.key === "ArrowLeft" ||
        event.key === "ArrowUp"
    ) {
        selectedSeat = (selectedSeat + 19) % 20;
    }

    else if (event.key.toLowerCase() === "b") {
        bookSeat();
        return;
    }

    else if (event.key.toLowerCase() === "v") {
        vacateSeat();
        return;
    }

    else if (event.key.toLowerCase() === "r") {
        resetSeats();
        return;
    }

    else {
        return;
    }

    updateSeats();
});

// 18. Animation and Rendering Loop
function animate() {

    requestAnimationFrame(animate);

    renderer.render(scene, camera);
}

// 19. Responsive Window Resize
new ResizeObserver(() => {

    const w = container.clientWidth;
    const h = container.clientHeight;

    if (!w || !h) {
        return;
    }

    camera.aspect = w / h;
    camera.updateProjectionMatrix();

    renderer.setSize(w, h);

}).observe(container);

// 20. Start the 3D Bus
updateSeats();
animate();
